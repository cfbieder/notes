## 2. Environment Separation (Dev vs Prod)

### Principle
- **Development**: Only the database runs in Docker. Backend and frontend run locally (fast HMR, instant reload).
- **Production**: Everything runs in Docker (DB, backend, nginx/frontend) as a self-contained stack.

### Environment Files

**backend/.env.example** (committed to git, template only):
```env
# Server
NODE_ENV=development
PORT=3000

# Database
DB_HOST=localhost
DB_PORT=5432
DB_NAME=myapp_dev
DB_USER=myappuser
DB_PASSWORD=dev_password_change_in_production

# Authentication
JWT_SECRET=CHANGE_ME
JWT_REFRESH_SECRET=CHANGE_ME
JWT_EXPIRATION=30m
JWT_REFRESH_EXPIRATION=7d

# Security
ENCRYPTION_KEY=CHANGE_ME

# File uploads
MAX_FILE_SIZE=10485760
UPLOAD_DIR=./uploads-dev

# Logging
LOG_LEVEL=debug
LOG_DIR=./logs

# CORS
CORS_ORIGIN=http://localhost:5173

# Rate Limiting
RATE_LIMIT_WINDOW_MS=900000
RATE_LIMIT_MAX_REQUESTS=100

# External API keys (optional)
# API_KEY=
```

**backend/.env.dev** (gitignored, auto-generated by setup-dev.sh):
- Same as .env.example but with real auto-generated secrets

**backend/.env** (gitignored, production values):
- `NODE_ENV=production`, `DB_HOST=postgres` (Docker service name), production secrets

**Environment loading strategy** (backend/server.js):
```javascript
require('dotenv').config({
  path: process.env.NODE_ENV === 'production' ? '.env' : '.env.dev'
});
```

**frontend/.env** (committed to git):
```env
VITE_APP_VERSION=0.1.0 (01.1.2026)
VITE_ENV_LABEL=DEV
```
- `VITE_ENV_LABEL=DEV` is present in the committed file for local development.
- In production, the Docker build does NOT pass `VITE_ENV_LABEL`, so it is undefined — making `Boolean(import.meta.env.VITE_ENV_LABEL)` return `false`.

### .gitignore entries
```gitignore

```

## 3. Docker Configuration

### docker-compose.dev.yml (Dev: DB only)
```yaml
services:
  postgres-dev:
    image: postgres:16-alpine
    container_name: myapp-db-dev
    restart: unless-stopped
    environment:
      POSTGRES_USER: myappuser
      POSTGRES_PASSWORD: dev_password_change_in_production
      POSTGRES_DB: myapp_dev
      POSTGRES_INITDB_ARGS: "--encoding=UTF8 --lc-collate=C --lc-ctype=C"
    ports:
      - "5432:5432"
    volumes:
      - postgres_dev_data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U myappuser -d myapp_dev"]
      interval: 10s
      timeout: 5s
      retries: 5
    networks:
      - myapp-network

volumes:
  postgres_dev_data:
    driver: local

networks:
  myapp-network:
    driver: bridge
```

### docker-compose.prod.yml (Prod: full stack)
```yaml
services:
  postgres:
    container_name: myapp-db
    image: postgres:16-alpine
    restart: unless-stopped
    environment:
      POSTGRES_USER: myappuser
      POSTGRES_PASSWORD: ${DB_PASSWORD}
      POSTGRES_DB: myapp_prod
      POSTGRES_INITDB_ARGS: "--encoding=UTF8 --lc-collate=C --lc-ctype=C"
    volumes:
      - postgres_prod_data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U myappuser -d myapp_prod"]
      interval: 10s
      timeout: 5s
      retries: 5
    networks:
      - myapp-prod-network

  backend:
    container_name: myapp-api
    build: ./backend
    restart: unless-stopped
    depends_on:
      postgres:
        condition: service_healthy
    env_file: ./backend/.env
    environment:
      NODE_ENV: production
      DB_HOST: postgres
      DB_PORT: 5432
      DB_NAME: myapp_prod
    volumes:
      - ./backend/uploads:/app/uploads
      - ./backend/logs:/app/logs
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:3000/health"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 30s
    networks:
      - myapp-prod-network

  nginx:
    container_name: myapp-web
    build:
      context: ./frontend
      args:
        VITE_APP_VERSION: "${APP_VERSION:-dev}"
        # NOTE: VITE_ENV_LABEL is intentionally NOT passed here
    restart: unless-stopped
    depends_on:
      backend:
        condition: service_healthy
    ports:
      - "443:443"
      - "80:80"
    volumes:
      - ./nginx/nginx.conf:/etc/nginx/nginx.conf:ro
      - ./certs:/etc/nginx/certs:ro
    networks:
      - myapp-prod-network

volumes:
  postgres_prod_data:

networks:
  myapp-prod-network:
    driver: bridge
```

### backend/Dockerfile
```dockerfile
# Stage 1: Install production dependencies
FROM node:20-alpine AS deps
WORKDIR /app
COPY package*.json ./
RUN npm ci --only=production

# Stage 2: Production image
FROM node:20-alpine
RUN apk add --no-cache curl
WORKDIR /app
RUN addgroup -g 1001 appgroup && adduser -u 1001 -G appgroup -D appuser
COPY --from=deps /app/node_modules ./node_modules
COPY . .
RUN mkdir -p uploads logs && chown -R appuser:appgroup uploads logs
USER appuser
EXPOSE 3000
HEALTHCHECK --interval=30s --timeout=10s --retries=3 \
  CMD curl -f http://localhost:3000/health || exit 1
CMD ["node", "server.js"]
```

### frontend/Dockerfile
```dockerfile
# Stage 1: Build
FROM node:20-alpine AS build
WORKDIR /app
COPY package*.json ./
RUN npm ci
COPY . .
ARG VITE_APP_VERSION=dev
ENV VITE_APP_VERSION=$VITE_APP_VERSION
# VITE_ENV_LABEL is intentionally NOT set — production has no label
RUN npm run build

# Stage 2: Serve with nginx
FROM nginx:alpine
COPY --from=build /app/dist /usr/share/nginx/html
RUN rm /etc/nginx/conf.d/default.conf
EXPOSE 80 443
CMD ["nginx", "-g", "daemon off;"]
```

### nginx/nginx.conf
```nginx
events {
    worker_connections 1024;
}

http {
    include       /etc/nginx/mime.types;
    default_type  application/octet-stream;

    # Redirect HTTP to HTTPS
    server {
        listen 80;
        return 301 https://$host$request_uri;
    }

    server {
        listen 443 ssl;

        ssl_certificate     /etc/nginx/certs/your-domain.crt;
        ssl_certificate_key /etc/nginx/certs/your-domain.key;
        ssl_protocols       TLSv1.2 TLSv1.3;

        client_max_body_size 10m;

        # Static assets — long cache
        location /assets/ {
            root /usr/share/nginx/html;
            expires 1y;
            add_header Cache-Control "public, immutable";
        }

        # SPA routing
        location / {
            root /usr/share/nginx/html;
            try_files $uri $uri/ /index.html;
        }

        # API reverse proxy
        location /api/ {
            proxy_pass http://backend:3000;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            proxy_read_timeout 300s;  # Long timeout for heavy operations
        }

        # Health check endpoint
        location = /health {
            proxy_pass http://backend:3000/health;
        }
    }
}
```

## 4. Database Migrations

### Strategy
- Numbered SQL files: `001_initial_schema.sql`, `002_add_users_table.sql`, etc.
- Migration runner tracks applied migrations in a `migrations_applied` table.
- Idempotent: use `IF NOT EXISTS`, `ON CONFLICT DO NOTHING` in migration SQL.
- Forward-only (no rollback in production — safer to write a new forward migration).
- Automatic `updated_at` triggers on tables that need them.

### Migration Runner (backend/src/utils/migrate.js)
The runner should:
1. Create `migrations_applied` table if it doesn't exist
2. Read all `.sql` files from `backend/migrations/`, sorted by name
3. Skip already-applied migrations
4. Execute each new migration in a transaction
5. Record the migration name after successful application

### package.json scripts
```json
{
  "scripts": {
    "migrate:dev": "node src/utils/migrate.js dev",
    "migrate:prod": "node src/utils/migrate.js prod",
    "migrate:create": "node src/utils/migrate.js create"
  }
}
```

### Important: Production Migration Gotcha
If your dev database is exposed on localhost:5432 and your production DB is inside Docker, `npm run migrate:prod` from the host will hit the dev DB. Apply production migrations via:
```bash
docker exec -i myapp-db psql -U myappuser -d myapp_prod < backend/migrations/NNN_name.sql
```
Or run the migration runner inside the production backend container:
```bash
docker exec myapp-api node src/utils/migrate.js prod
```

## 5. Scripts

All scripts live in `scripts/` and follow these conventions:
- `set -euo pipefail` for safety
- Color-coded output (RED/GREEN/YELLOW/BLUE)
- `--help` flag on every script
- Auto-detect project root: `PROJECT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"`

### 5.1 setup-dev.sh — One-Command Dev Setup

Run once after a fresh clone to bootstrap the dev environment.

**What it does:**
1. Installs Node.js 20.x (idempotent — skips if already present)
2. Installs Docker + Compose plugin (idempotent)
3. Creates `backend/.env.dev` from `.env.example` with auto-generated secrets (`openssl rand -hex 32` for JWT secrets, etc.)
4. Runs `npm install` for backend and frontend
5. Starts dev PostgreSQL container
6. Runs database migrations
7. Prints instructions for starting dev servers

```bash
#!/usr/bin/env bash
set -euo pipefail

RED='\033[0;31m'; GREEN='\033[0;32m'
YELLOW='\033[1;33m'; BLUE='\033[0;34m'; NC='\033[0m'

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
PROJECT_ROOT="$(cd "$SCRIPT_DIR/.." && pwd)"

log()  { echo -e "${BLUE}[setup]${NC} $*"; }
ok()   { echo -e "${GREEN}[ok]${NC} $*"; }
warn() { echo -e "${YELLOW}[warn]${NC} $*"; }
die()  { echo -e "${RED}[error]${NC} $*" >&2; exit 1; }

# 1. Install Node.js 20
install_node() {
  if command -v node &>/dev/null && [[ "$(node --version)" == v20* ]]; then
    ok "Node.js $(node --version) already installed"; return
  fi
  log "Installing Node.js 20.x ..."
  curl -fsSL https://deb.nodesource.com/setup_20.x | sudo -E bash -
  sudo apt-get install -y nodejs
  ok "Node.js $(node --version) installed"
}

# 2. Install Docker
install_docker() {
  if command -v docker &>/dev/null; then
    ok "Docker already installed"; return
  fi
  log "Installing Docker ..."
  sudo apt-get update -qq
  sudo apt-get install -y ca-certificates curl gnupg lsb-release
  sudo install -m 0755 -d /etc/apt/keyrings
  curl -fsSL https://download.docker.com/linux/ubuntu/gpg \
    | sudo gpg --dearmor -o /etc/apt/keyrings/docker.gpg
  echo "deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.gpg] \
    https://download.docker.com/linux/ubuntu $(lsb_release -cs) stable" \
    | sudo tee /etc/apt/sources.list.d/docker.list > /dev/null
  sudo apt-get update -qq
  sudo apt-get install -y docker-ce docker-ce-cli containerd.io docker-compose-plugin
  # Add user to docker group
  if ! groups "$USER" | grep -q docker; then
    sudo usermod -aG docker "$USER"
    warn "Log out and back in for docker group to take effect"
  fi
  ok "Docker installed"
}

# 3. Create .env.dev from template
create_env() {
  local env_file="$PROJECT_ROOT/backend/.env.dev"
  if [[ -f "$env_file" ]]; then ok ".env.dev already exists"; return; fi
  log "Creating backend/.env.dev from .env.example ..."
  cp "$PROJECT_ROOT/backend/.env.example" "$env_file"
  sed -i "s|JWT_SECRET=.*|JWT_SECRET=$(openssl rand -hex 32)|" "$env_file"
  sed -i "s|JWT_REFRESH_SECRET=.*|JWT_REFRESH_SECRET=$(openssl rand -hex 32)|" "$env_file"
  sed -i "s|ENCRYPTION_KEY=.*|ENCRYPTION_KEY=$(openssl rand -hex 16)|" "$env_file"
  ok ".env.dev created with auto-generated secrets"
}

# 4. Install dependencies
install_deps() {
  (cd "$PROJECT_ROOT/backend" && npm install)
  (cd "$PROJECT_ROOT/frontend" && npm install)
  ok "Dependencies installed"
}

# 5. Start dev DB
start_db() {
  docker compose -f "$PROJECT_ROOT/docker-compose.dev.yml" up -d
  local retries=30
  until docker exec myapp-db-dev pg_isready -U myappuser -d myapp_dev &>/dev/null; do
    retries=$((retries - 1))
    [[ $retries -eq 0 ]] && die "PostgreSQL did not become ready"
    sleep 1
  done
  ok "PostgreSQL is ready"
}

# 6. Run migrations
run_migrations() {
  (cd "$PROJECT_ROOT/backend" && npm run migrate:dev)
  ok "Migrations applied"
}

main() {
  install_node
  install_docker
  create_env
  install_deps
  start_db
  run_migrations
  echo ""
  ok "Setup complete!"
  echo "Start dev servers:"
  echo "  Terminal 1: cd backend && npm run dev"
  echo "  Terminal 2: cd frontend && npm run dev"
}
main "$@"
```

### 5.2 deploy-to-production.sh

Full production deployment with pre-flight checks, automatic backup, Docker build, migrations, and health verification.

**Flags:** `--with-git` (pull before deploy), `--no-backup` (skip DB backup, for first deploy)

**Flow:**
1. Verify docker-compose.prod.yml, backend/.env, SSL certs exist
2. Optional `git pull`
3. Backup existing production database (pg_dump -Fc)
4. Read app version from frontend/.env, export for Docker build arg
5. `docker compose -f docker-compose.prod.yml build --no-cache`
6. `docker compose -f docker-compose.prod.yml up -d`
7. Wait for DB readiness (pg_isready loop, 30 retries x 2s)
8. Run migrations inside backend container: `docker exec myapp-api node src/utils/migrate.js prod`
9. Wait for backend health check
10. Verify all containers running + HTTPS endpoint responds

```bash
#!/bin/bash
# deploy-to-production.sh - Deploy to production
# Usage: ./scripts/deploy-to-production.sh [--with-git] [--no-backup] [--help]

set -e

RED='\033[0;31m'; GREEN='\033[0;32m'
YELLOW='\033[1;33m'; BLUE='\033[0;34m'; NC='\033[0m'

PROJECT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
COMPOSE_FILE="$PROJECT_DIR/docker-compose.prod.yml"
BACKUP_DIR="$PROJECT_DIR/Backups"
PROD_URL="https://your-domain.example.com"

WITH_GIT=false; NO_BACKUP=false

while [[ $# -gt 0 ]]; do
    case $1 in
        --with-git)  WITH_GIT=true; shift ;;
        --no-backup) NO_BACKUP=true; shift ;;
        --help)
            echo "Usage: $0 [--with-git] [--no-backup] [--help]"; exit 0 ;;
        *) echo -e "${RED}Unknown option: $1${NC}"; exit 1 ;;
    esac
done

echo -e "${BLUE}========================================${NC}"
echo -e "${BLUE}  Production Deploy${NC}"
echo -e "${BLUE}========================================${NC}"
cd "$PROJECT_DIR"

# Pre-flight checks
[ ! -f "$COMPOSE_FILE" ] && echo -e "${RED}ERROR: docker-compose.prod.yml not found${NC}" && exit 1
[ ! -f "$PROJECT_DIR/backend/.env" ] && echo -e "${RED}ERROR: backend/.env not found${NC}" && exit 1
echo -e "${GREEN}✓ Config files verified${NC}"

# Optional git pull
[ "$WITH_GIT" = true ] && git pull && echo -e "${GREEN}✓ Git pull complete${NC}"

# Backup existing DB
if [ "$NO_BACKUP" = false ]; then
    if docker ps --format '{{.Names}}' | grep -q '^myapp-db$'; then
        mkdir -p "$BACKUP_DIR"
        TIMESTAMP=$(date +%Y%m%d_%H%M%S)
        BACKUP_FILE="$BACKUP_DIR/backup_${TIMESTAMP}.dump"
        docker exec myapp-db pg_dump -U myappuser -d myapp_prod -Fc > "$BACKUP_FILE"
        echo -e "${GREEN}✓ DB backed up: $BACKUP_FILE ($(du -h "$BACKUP_FILE" | cut -f1))${NC}"
    fi
fi

# Get version for build arg
APP_VERSION="dev"
[ -f "$PROJECT_DIR/frontend/.env" ] && APP_VERSION=$(grep VITE_APP_VERSION "$PROJECT_DIR/frontend/.env" | cut -d'=' -f2)
export APP_VERSION
[ -f "$PROJECT_DIR/backend/.env" ] && export DB_PASSWORD=$(grep '^DB_PASSWORD=' "$PROJECT_DIR/backend/.env" | sed 's/^DB_PASSWORD=//')

# Build and start
echo -e "${YELLOW}Building Docker images (version: $APP_VERSION)...${NC}"
docker compose -f "$COMPOSE_FILE" build --no-cache
docker compose -f "$COMPOSE_FILE" up -d

# Wait for DB
for i in $(seq 1 30); do
    docker exec myapp-db pg_isready -U myappuser -d myapp_prod > /dev/null 2>&1 && break
    [ $i -eq 30 ] && echo -e "${RED}ERROR: DB not ready${NC}" && exit 1
    sleep 2
done
echo -e "${GREEN}✓ Database ready${NC}"

# Run migrations
docker exec myapp-api node src/utils/migrate.js prod
echo -e "${GREEN}✓ Migrations complete${NC}"

# Verify
CONTAINERS=("myapp-db" "myapp-api" "myapp-web")
ALL_RUNNING=true
for c in "${CONTAINERS[@]}"; do
    docker ps --format '{{.Names}}' | grep -q "^${c}$" || ALL_RUNNING=false
done

echo -e "${BLUE}========================================${NC}"
[ "$ALL_RUNNING" = true ] && echo -e "${GREEN}  Deployment successful! URL: $PROD_URL${NC}" \
    || echo -e "${RED}  Issues detected — check: docker logs myapp-api${NC}"
echo -e "${BLUE}========================================${NC}"
```

### 5.3 update_version.sh — Semantic Versioning with Date

Manages the app version string stored in `frontend/.env`. Format: `MAJOR.MINOR.PATCH (DD.M.YYYY)`.

**Modes:**
- `--show` — Display current version
- `--auto` — Update date only, keep version number
- `--bump-patch` — 1.2.3 -> 1.2.4
- `--bump-minor` — 1.2.3 -> 1.3.0
- `--bump-major` — 1.2.3 -> 2.0.0
- `"1.2.0 (30.1.2026)"` — Set exact version string

**Also creates a git tag** (`v1.2.0`) automatically.

```bash
#!/bin/bash
# update_version.sh - Update app version
# Usage: ./scripts/update_version.sh --bump-patch | --bump-minor | --bump-major | --auto | --show | "VERSION"

set -euo pipefail

ENV_FILE="frontend/.env"
VERSION_KEY="VITE_APP_VERSION"

# Ensure we're in project root
if [ ! -f "$ENV_FILE" ]; then
  [ -f "../$ENV_FILE" ] && cd .. || { echo "Error: Cannot find $ENV_FILE"; exit 1; }
fi

get_current_version() { grep "^${VERSION_KEY}=" "$ENV_FILE" | cut -d'=' -f2-; }
extract_version_number() { echo "$1" | grep -oE '[0-9]+\.[0-9]+\.[0-9]+' | head -1; }
get_date_string() { date +"%d.%-m.%Y"; }

update_version() {
  if grep -q "^${VERSION_KEY}=" "$ENV_FILE"; then
    sed -i "s/^${VERSION_KEY}=.*/${VERSION_KEY}=${1}/" "$ENV_FILE"
  else
    echo "${VERSION_KEY}=${1}" >> "$ENV_FILE"
  fi
}

CURRENT_VERSION=$(get_current_version)
CURRENT_NUM=$(extract_version_number "$CURRENT_VERSION")
DATE_STR=$(get_date_string)

case "${1:---help}" in
  --show)   echo "Current version: $CURRENT_VERSION"; exit 0 ;;
  --auto)   NEW_VERSION="${CURRENT_NUM} (${DATE_STR})" ;;
  --bump-patch)
    IFS='.' read -r major minor patch <<< "$CURRENT_NUM"
    NEW_VERSION="${major}.${minor}.$((patch + 1)) (${DATE_STR})" ;;
  --bump-minor)
    IFS='.' read -r major minor patch <<< "$CURRENT_NUM"
    NEW_VERSION="${major}.$((minor + 1)).0 (${DATE_STR})" ;;
  --bump-major)
    IFS='.' read -r major minor patch <<< "$CURRENT_NUM"
    NEW_VERSION="$((major + 1)).0.0 (${DATE_STR})" ;;
  --help|-h)
    echo "Usage: $0 --bump-patch | --bump-minor | --bump-major | --auto | --show | \"VERSION\""
    echo "Current: $(get_current_version)"; exit 0 ;;
  --*)  echo "Unknown option: $1"; exit 1 ;;
  *)    NEW_VERSION="$1" ;;
esac

echo "Updating version: $CURRENT_VERSION -> $NEW_VERSION"
update_version "$NEW_VERSION"

# Create git tag
TAG_NAME="v${NEW_VERSION%% *}"
if git rev-parse "$TAG_NAME" >/dev/null 2>&1; then
  echo "Git tag '$TAG_NAME' already exists, skipping."
else
  git tag -a "$TAG_NAME" -m "Version ${NEW_VERSION}"
  echo "Git tag created: $TAG_NAME (push with: git push origin $TAG_NAME)"
fi

echo "Rebuild frontend for changes to take effect: cd frontend && npm run build"
```

### 5.4 backup-db.sh — Local Database Backup

```bash
#!/bin/bash
# backup-db.sh - Backup production database
# Usage: ./scripts/backup-db.sh [--keep N]

set -e

RED='\033[0;31m'; GREEN='\033[0;32m'
YELLOW='\033[1;33m'; BLUE='\033[0;34m'; NC='\033[0m'

PROJECT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
BACKUP_DIR="$PROJECT_DIR/Backups"
CONTAINER="myapp-db"
DB_NAME="myapp_prod"
DB_USER="myappuser"
KEEP_BACKUPS=0  # 0 = keep all

while [[ $# -gt 0 ]]; do
    case $1 in
        --keep) KEEP_BACKUPS="$2"; shift 2 ;;
        --help) echo "Usage: $0 [--keep N]"; exit 0 ;;
        *) echo -e "${RED}Unknown: $1${NC}"; exit 1 ;;
    esac
done

# Check container running
docker ps --format '{{.Names}}' | grep -q "^${CONTAINER}$" \
    || { echo -e "${RED}ERROR: '${CONTAINER}' not running${NC}"; exit 1; }

mkdir -p "$BACKUP_DIR"
TIMESTAMP=$(date +%Y%m%d_%H%M%S)
BACKUP_FILE="$BACKUP_DIR/backup_${TIMESTAMP}.dump"

echo -e "${YELLOW}Backing up ${DB_NAME}...${NC}"
docker exec "$CONTAINER" pg_dump -U "$DB_USER" -d "$DB_NAME" -Fc > "$BACKUP_FILE"
echo -e "${GREEN}✓ Backup: $BACKUP_FILE ($(du -h "$BACKUP_FILE" | cut -f1))${NC}"

# Prune old backups
if [ "$KEEP_BACKUPS" -gt 0 ]; then
    COUNT=$(ls -1 "$BACKUP_DIR"/backup_*.dump 2>/dev/null | wc -l)
    if [ "$COUNT" -gt "$KEEP_BACKUPS" ]; then
        REMOVE=$((COUNT - KEEP_BACKUPS))
        echo -e "${YELLOW}Pruning $REMOVE old backup(s)...${NC}"
        ls -1t "$BACKUP_DIR"/backup_*.dump | tail -n "$REMOVE" | xargs rm -f
    fi
fi

echo -e "${BLUE}Current backups:${NC}"
ls -lh "$BACKUP_DIR"/backup_*.dump 2>/dev/null | awk '{print "  " $NF " (" $5 ")"}'
```

### 5.5 backup-to-remote.sh — Remote Backup via SSH

Comprehensive backup script that:
1. Checks SSH connectivity to remote host
2. Detects running database container (prod, falls back to dev)
3. Dumps database with `pg_dump -Fc`
4. Copies environment files and data directories
5. Creates tarball and transfers via SCP
6. Cleans remote backups older than retention period (default 30 days)
7. Verifies transfer with remote file count

**Features:** `--dry-run` flag, cleanup trap, dual logging (stdout + file), configurable retention.

**Configuration block at top of script:**
```bash
REMOTE_HOST="192.168.1.252"
REMOTE_USER="youruser"
REMOTE_DIR="~/backups/myapp"
DB_CONTAINER_PROD="myapp-db"
DB_CONTAINER_DEV="myapp-db-dev"
DB_USER="myappuser"
DB_NAME="myapp_prod"
RETENTION_DAYS=30
BACKUP_DIRS=("backend/uploads" "Backups")
ENV_FILES=("backend/.env" "backend/.env.dev")
```

### 5.6 sync-db-prod-to-dev.sh — Production to Dev Database Sync

```bash
#!/bin/bash
# sync-db-prod-to-dev.sh - Copy production database to development
# Usage: ./scripts/sync-db-prod-to-dev.sh [--with-uploads]

set -e

RED='\033[0;31m'; GREEN='\033[0;32m'
YELLOW='\033[1;33m'; BLUE='\033[0;34m'; NC='\033[0m'

PROJECT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
PROD_CONTAINER="myapp-db"
DEV_CONTAINER="myapp-db-dev"
PROD_DB="myapp_prod"
DEV_DB="myapp_dev"
DB_USER="myappuser"
TEMP_DUMP="/tmp/myapp_prod_dump.dump"

echo -e "${BLUE}  Sync Production DB -> Development${NC}"

# Check prod container
docker ps --format '{{.Names}}' | grep -q "^${PROD_CONTAINER}$" \
    || { echo -e "${RED}ERROR: '${PROD_CONTAINER}' not running${NC}"; exit 1; }

# Start dev container if needed
if ! docker ps --format '{{.Names}}' | grep -q "^${DEV_CONTAINER}$"; then
    docker compose -f "$PROJECT_DIR/docker-compose.dev.yml" up -d postgres
    for i in $(seq 1 15); do
        docker exec "$DEV_CONTAINER" pg_isready -U "$DB_USER" -d "$DEV_DB" > /dev/null 2>&1 && break
        sleep 2
    done
fi

# Show current counts
echo -e "${BLUE}Production database contents:${NC}"
for table in users documents; do
    COUNT=$(docker exec "$PROD_CONTAINER" psql -U "$DB_USER" -d "$PROD_DB" -t -c \
        "SELECT COUNT(*) FROM $table;" 2>/dev/null | tr -d ' ' || echo "N/A")
    echo "  $table: $COUNT rows"
done

# Confirm
echo -e "${RED}WARNING: This will REPLACE all data in dev!${NC}"
read -p "Continue? (y/N): " CONFIRM
[[ ! "$CONFIRM" =~ ^[Yy]$ ]] && exit 0

# Dump and restore
docker exec "$PROD_CONTAINER" pg_dump -U "$DB_USER" -d "$PROD_DB" -Fc > "$TEMP_DUMP"
echo -e "${GREEN}✓ Dump created ($(du -h "$TEMP_DUMP" | cut -f1))${NC}"

cat "$TEMP_DUMP" | docker exec -i "$DEV_CONTAINER" pg_restore \
    -U "$DB_USER" -d "$DEV_DB" --clean --if-exists --no-owner 2>/dev/null || true
echo -e "${GREEN}✓ Database restored to dev${NC}"

rm -f "$TEMP_DUMP"
echo -e "${GREEN}✓ Sync complete!${NC}"
```

## 8. Backup Strategy

### Three Tiers

1. **Pre-deploy backup** — Automatic before every production deployment (in deploy-to-production.sh)
2. **Local on-demand backup** — `./scripts/backup-db.sh` with optional pruning
3. **Remote automated backup** — `./scripts/backup-to-remote.sh` on a cron schedule

### Cron Jobs

```cron
# Remote backup every 2 days at 2 AM
0 2 */2 * * /path/to/project/scripts/backup-to-remote.sh >> /path/to/project/Backups/backup-remote.log 2>&1

# TLS certificate renewal (monthly)
0 3 1 * * cd /path/to/project && ./scripts/setup-certs.sh && docker exec myapp-web nginx -s reload

# Docker disk cleanup (weekly)
0 3 * * 0 docker builder prune --force --filter until=48h && docker image prune --force --filter until=48h >> /path/to/project/logs/docker-prune.log 2>&1
```

### Backup Contents
- **Database**: `pg_dump -Fc` (custom format, compressed, supports selective restore)
- **Upload files**: User-uploaded files in `backend/uploads/`
- **Environment files**: `.env`, `.env.dev` (contain encrypted API keys)

### Retention
- Local: All backups kept (or use `--keep N` to prune)
- Remote: Configurable (default 30 days), auto-cleaned by `find -mtime`

## 13. Additional Best Practices

### Health Check Endpoint
Every backend should expose `/health` for Docker health checks and monitoring:
```javascript
app.get('/health', (req, res) => {
  res.json({ status: 'ok', timestamp: new Date().toISOString() });
});
```

### Non-Root Docker User
Always run as a non-root user in production containers:
```dockerfile
RUN addgroup -g 1001 appgroup && adduser -u 1001 -G appgroup -D appuser
USER appuser
```

### JWT Auth with Refresh Tokens
- Access token: Short-lived (30 minutes)
- Refresh token: Longer-lived (7 days), stored in httpOnly cookie
- Rotate refresh tokens on use for security

### Rate Limiting
Apply rate limiting to API endpoints, especially auth:
```javascript
const rateLimit = require('express-rate-limit');
app.use('/api/', rateLimit({
  windowMs: 15 * 60 * 1000,  // 15 minutes
  max: 100,
}));
```

### Helmet for Security Headers
```javascript
const helmet = require('helmet');
app.use(helmet());
```

### Structured Logging with Winston
Use Winston for production logging with file rotation:
```javascript
const winston = require('winston');
const logger = winston.createLogger({
  level: process.env.LOG_LEVEL || 'info',
  format: winston.format.combine(
    winston.format.timestamp(),
    winston.format.json()
  ),
  transports: [
    new winston.transports.File({ filename: 'logs/error.log', level: 'error' }),
    new winston.transports.File({ filename: 'logs/combined.log' }),
  ],
});
```

### Input Validation with Joi
Validate all API inputs at the boundary:
```javascript
const Joi = require('joi');
const schema = Joi.object({
  email: Joi.string().email().required(),
  password: Joi.string().min(8).required(),
});
```

### User-Partitioned Uploads
Store uploads in user-specific directories to avoid conflicts and simplify cleanup:
```
uploads/
  <user_id>/
    2026/
      01/
        file1.pdf
```

### Docker Disk Cleanup
Docker images accumulate over time. Schedule weekly cleanup:
```bash
docker builder prune --force --filter until=48h
docker image prune --force --filter until=48h
```

### Port Conflict Resolution
Dev servers sometimes leave orphaned processes:
```bash
sudo kill -9 $(sudo lsof -t -i:3000)
sudo kill -9 $(sudo lsof -t -i:5173)
```

### Engine Requirements
Specify in package.json to catch version mismatches early:
```json
{
  "engines": {
    "node": ">=18.0.0",
    "npm": ">=9.0.0"
  }
}
```
